public class ListaDE<E> implements PositionList<E>{

  protected Nodo<E> head;
  protected Nodo<E> tail;
  protected int cantElems;

  public int duplicarElem(E elem) throws EmptyListException{
    if(this.isEmpty()){
	throw new EmptyListException("La lista está vacía");
    }
    int cont = 0;
    Position<E> p = this.next(head);
    while(p != tail){
      if(p.element().equals(elem)){
	this.addAfter(p, elem);
	cont++;
        p = this.next(p);
      }
      p = this.next(p);
    }
    return cont;
  }

  public boolean isEmpty(){
  return cantElems == 0;
  }
  public Position<E> next(Position <E> p) throws InvalidPositionException{
    if(this.isEmpty() || p == null || !(p instanceof Nodo<?>) || p.element() == null){ //Asumo que no se pueden insertar nodos vacíos
      throw new InvalidPositionException("La posición es inválida");
    }
    return (Nodo<E>)p.getNext();
  }
  public void addAfter(Position<E> p, E elem){
    if(this.isEmpty() || p == null || !(p instanceof Nodo<?>) || p.element() == null){ //Asumo que no se pueden insertar nodos vacíos
      throw new InvalidPositionException("La posición es inválida");
    }
    Nodo<E> n = new Nodo<E>(elem);
    Nodo<E> siguiente = (Nodo<E>)this.next((Nodo<E>)p);
    n.setPrev((Nodo<E>)p);
    n.setNext(siguiente);
    ((Nodo<E>)p).setNext(n);
    siguiente.setPrev(n);
    cantElems++;
  }
}

public Stack<Character> eliminarDePila (Stack<Character> s, char ch){
  Stack<Character> aux = new TDAPila<>();
  Stack<Character> p = new TDAPila<>();

  while (!(s.isEmpty())){
    char temp = s.pop();
      if(temp != ch){
	aux.push(temp);
      }
  }
  while(!(aux.isEmpty())){
    p.push(aux.pop());
  }
  return p;
}
