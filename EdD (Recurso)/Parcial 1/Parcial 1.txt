	pos.getPrev.setNext(aux.getNext());
	pos.getNext.setPrev(aux.getPrev);
	cantElements--;
	return pos.element();
   }
   
}

public void insertar(Stack<E> p, Queue<E> q){
   Pila<E> aux = new Pila<E>();
   while(!(p.isEmpty())){
	aux.push(p.pop());
   }
   while(!(q.isEmpty())){
	p.push(q.dequeue());
   }
   while(!(aux.isEmpty())){
	p.push(aux.pop());
   }
}

public Dictionary<Character, Position<Character>> caracteresYposiciones(PositionList<Character> l){
   Dictionary<Character, Position<Character>> diccionario = new Dictionary<Character, Position<Character>>();
   for(Position<Character> p : l.positions()){
	diccionario.insert(p.element(), p);
   }
   return diccionario;
}
