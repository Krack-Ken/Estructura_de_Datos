public class DiccionarioHashAbierto<K,V> implements Dictionary<K,V>{
    protected satic final float fc = o.7f;
    protected positionList<Entrada<K,V>>[] buckets;
    protected int n;
    protected int N;

    public boolean kEnV(K k, V v) throws InvalidKeyException{
        if(k == null){
            throw new InvalidKeyException("La llave no puede ser nula")
        }
        if(find(k) == null){
            throw new InvalidKeyException("La llave no pertenece al diccionario")
        }
        int balde = Math.abs(k.hashCode()) % N;
        boolean aux = !buckets[balde].isEmpty();
        for(Entry<K, V> e : buckets[balde]){
            if(e.getKey().equals(k) && !e.getValue().equals(v)){
                aux = false;
                break;
            }
        }
        return aux;
    }

    public Entry<K,V> find(K key) throws InvalidKeyException{
        if(key == null){
            throw new InvalidKeyException("La llave no puede ser nula")
        }
        int balde = Math.abs(key.hashCode()) % N;
        for(Entry<K,V> e : buckets[balde]){
            if (e.getKey().equals(key)){
                return e;
            }
        }
        return null;
    }
}

public class Arbol<E> implements Tree<E>{
    protected Tnodo<E> raiz;
    protected int cant;

    public Iterable<E> eliminarDescendientes(Position<E> p) throws InvalidPositionException{
        if (p == null || !(p instanceof TNodo<E>)){
            throw new InvalidPositionException("la posición no puede ser nula");
        }
        TNodo<E> subRaiz = (TNodo<E>)p;
        PositionList<E> lista = new TDALista<E>();
        for(Position<TNodo<E>> Pn : subRaiz.getHijos().positions()){
            eliminarRecursivo(Pn, lista);
        }
        return lista;
    }
    private void eliminarRecursivo(Position<TNodo<E>> Pn, PositionList<E> lista){
        TNodo<E> nodo = Pn.element();
        if(!nodo.getHijos().isEmpty()){
            //Me aseguro de que eliminar primero hojas
            for(Position<TNodo<E>> Pn2 : Pn.element().getHijos().positions()){
                eliminarRecursivo(Pn2, lista);
            }
        }
        lista.addLast(nodo.element());
        nodo.getPadre().getHijos().remove(Pn);
        nodo.setPadre(null);
        nodo.setElement(null);
        cant--;
    }
}

a) Escriba un método que reciba un arbol binario que representa una expresión aritmética bien formada compuesta únicamente por los operadores +, -, *, y /
El método debe retornar un mapeo cuyas claves sean:
    "Pares"
    "Impares"
    "Operadores"
y cuyos valores indiquen respectivamente la cantidad de cada uno almacenado en el árbol
Para resolver este ejercicio deben implementar un único recorrido inorden y sin utilizar los iteradores del arbol (iterator ni position).
Asuma que cuenta con los TDA ArbolBinario y Mapeo totalmente implementados. Si utiliza métodos auxiliares debe implementarlos

//Asumo que el árbol es un Binary<Object>
public Mapeo<String, Integer> deArbolAMapa(BinaryTree<Object> bt){
    Mapeo<String, Integer> mp = new Mapeo<String, Integer>();
    mp.put("Pares", 0);
    mp.put("Impares", 0);
    mp.put("Operadores", 0);
    if(!bt.isEmpty){
        recorrer(mp, bt, (BNodo<Object>)bt.root());
    }
    return mp;
}

public void recorrer(Mapeo<String, Integer> mp, BinaryTree<Object> bt, BNodo<Object> bn){
    if(bt.hasLeft(bn)){
        recorrer(mp, bt, (BNodo<Object>)bt.left(bn));
    }
    Object elem = bn.element();
    if(elem instanceof Integer){
        Integer e = (Integer)elem;
        if (e % 2 == 0){
            mp.put("Pares", mp.get("Pares")+1);
        }
        else{
            mp.put("Impares", mp.get("Impares")+1);
        }
    }
    else{ //Como el arbol solo contiene número o símbolos, si elem no es una instancia de un número, por descarte es un operador
        mp.put("Operadores", mp.get("Operadores")+1);
    }
    if(bt.hasRight(bn)){
        recorrer(mp, bt, (BNodo<Object>)bt.right(bn));
    }
}
