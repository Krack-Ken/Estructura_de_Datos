public class DiccionarioHashAbierto<K,V> implements Dictionary<K,V> {
    protected static final float fc = 0.7f;
    protected PositionList<Entrada<K,V>>[] buckets;
    protected int n;
    protected int N;
    public boolean almenosE(K key, int e) throws invalidKeyException{
        if(key == null){
            throw new invalidKeyException("La llave no puede ser nula");
        }
        int cont = 0;
        boolean aux = e <= 0; //si e < 0 o e == 0 siempre es verdadero
        int b = Math.abs(key.hashCode() % N);
        for(Entrada<K,V> en : buckets[b]){
            if(en.getKey().equals(key)){
                cont++;
            }
            if(cont >= e)
                return true;
        }
        return aux;
    }
}

Ejercicio 2:
Agregue un método a la clase Arbol<E> con la siguiente signatura: public Iterable<E> convertirAhoja(Position<E> p) throws InvalidPositionException tal que elimine del árbol receptor del mensaje todos los nodos que sean necesarios para convertir la posición p en hoja. public class Arbol<E> implements Tree<E>{
    protected Tnodo<E> raiz;
    protected int cant;

    public Iterable<E> convertirAhoja(Position<E> p) throws InvalidPositionException{
        if(p == null || !(p instanceof TNodo<E>)){
            throw new InvalidPositionException("La posición debe ser un nodo no nulo");
        }
        PositionList<E> l = new TDALista<E>();
        TNodo<E> tn = (TNodo<E>)p;
        if(!tn.getHijos().isEmpty()){
        //Asumo que se pueden eliminar elementos de la lista mientras la recorrés con el for
            for(Position<TNodo<E>> n = tn.getHijos().positions()){
                eliminarRecursivo(n, l);
            }
        }
        return l;
    }

    private void eliminarRecursivo(Position<TNodo<E>> n, PositionList<E> l){
        TNodo<E> tn = n.element();
        if(!tn.getHijos().isEmpty()){
        //Asumo que se pueden eliminar elementos de la lista mientras la recorrés con el for
            for(TNodo<E> n = tn.getHijos()){
                eliminarRecursivo(n, l);
            }
        }
        tn.getPadre().getHijos().remove(n);
        tn.setPadre(null);
        l.addLast(tn.element());
        tn.setElement(null);
        cant--;
    }
}

Ejercicio 3.
Escriba un método que reciba un árbol de caracteres y retorne un mapeo cuyas claves sean cada una de las vocales y sus resectivos valores la cantidad de veces que aparece cada vocal en el árbol. Para resolver este ejercicio debe implementar un único recorrido en posorden y sin usar los iteradores del árbol (ni iterator, ni positions). Considere solo letras en minúscula. Asuma que cuenta con los TDAs Arbol y Mapeo totalmnte implementados. Si utiliza métodos auxiliares debe implementarlos.

//Asumo que al menos el árbol contiene la raiz, es decir a.root() != null
public Mapeo<Character, Integer> cantVocales (Arbol<Character> a){
    TNodo<Character> raiz = (TNodo<Character>)a.root();
    Mapeo<Character, Integer> mapa = new Mapeo<Character, Integer>();
    mapa.put('a', 0);
    mapa.put('e', 0);
    mapa.put('i', 0);
    mapa.put('o', 0);
    mapa.put('u', 0);
    recorrerRecursivo(a, mapa, raiz);
    return mapa;
}

private void recorrerRecursivo(Arbol<Character> a, Mapeo<Character, Integer> mapa, TNodo<Character> t){
    if(!t.getHijos().isEmpty()){
        for(TNodo<Character> e : t.getHijos()){
            recorrerRecursivo(a, mapa, e);
        }
    }
    char letra = t.element();
    if(letra == 'a'){
        mapa.put('a', mapa.get('a')+1);
    }
    else if(letra == 'e'){
        mapa.put('e', mapa.get('e')+1);
    }
    else if(letra == 'i'){
        mapa.put('i', mapa.get('i')+1);
    }
    else if(letra == 'o'){
        mapa.put('o', mapa.get('o')+1);
    }
    else if(letra == 'u'){
        mapa.put('u', mapa.get('u')+1);
    }
}
